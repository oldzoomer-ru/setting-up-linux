### 11. late_command: NM fix, шрифты Microsoft, unrar ###
d-i preseed/late_command string \
  in-target apt-get update; \
  # --- NetworkManager fix ---
  # Удаляем из /etc/network/interfaces записи о физических Ethernet и Wi-Fi интерфейсах,
  # чтобы ими управлял только NetworkManager (ifupdown оставляет лишь loopback).
  in-target sh -c 'awk '' \
    /^[[:space:]]*#/ { next } \
    /^[[:space:]]*$/ { next } \
    { \
      if ($1 == "auto" || $1 == "allow-hotplug") { \
        if ($2 ~ /^(eth|en|wl|wlan|wlp|wlan[0-9]|en[opsx])/) next; \
        print; next; \
      } \
      if ($1 == "iface") { \
        if ($2 ~ /^(eth|en|wl|wlan|wlp|wlan[0-9]|en[opsx])/) next; \
        print; next; \
      } \
      print; \
    }'' /etc/network/interfaces > /etc/network/interfaces.new && \
    mv /etc/network/interfaces.new /etc/network/interfaces'; \
  # --- Шрифты Microsoft и unrar ---
  echo "ttf-mscorefonts-installer msttcorefonts/accepted-mscorefonts-eula select true" | in-target debconf-set-selections; \
  in-target apt-get install -y ttf-mscorefonts-installer unrar; \
  # --- Post-install configuration ---
  # Run first-boot initialization script (same as autoinstall.yaml postinstall)
  /target/bin/bash /root/first-boot-setup.sh || true
  in-target apt-get install -y firmware-linux firmware-linux-nonfree || true
